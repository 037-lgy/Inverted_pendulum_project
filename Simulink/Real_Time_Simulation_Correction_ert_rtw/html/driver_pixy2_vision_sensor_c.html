11#define COLORTRACKINGFRAMEWIDTH 315



12#define COLORTRACKINGFRAMEHEIGHT 207



13#define BYTESPERDETECTEDBLOCK 14



14#define LINETRACKINGFRAMEWIDTH 78



15#define LINETRACKINGFRAMEHEIGHT 51



16#define UPPERSYNCBYTE 193



17#define LOWERSYNCBYTE 175



18#define COLORTRACKINGPACKETTYPE 33



19#define LINETRACKINGPACKETTYPE 49



20



21static int iicFile = -1;



22IIC *pIic;



23static IICDAT iic_dat;



24



25void setIICCommand(uint8_T portNumber, uint8_T i2cAddress, uint8_T mode, uint8_T sampleTime, uint8_T colorSignature) 



26{



27    // Common i2c dat settings for all requests



28    iic_dat.Port = portNumber;



29    iic_dat.Time = sampleTime;



30    iic_dat.WrData[0] = i2cAddress;



31    iic_dat.WrData[1] = 00; // In case of pixy2, the firmware has knowledge of register to fetch data from, so send 00



32    iic_dat.WrData[2] = 0xAE;



33    iic_dat.WrData[3] = 0xC1;



34



35    if((mode == 0)||(mode == 1)||(mode == 2)) 



36    {



37        // Color Signature Tracking



38        iic_dat.RdLng = 0x14; 



39        iic_dat.WrLng = 0x08;  



40        iic_dat.WrData[4] = 0x20;



41        iic_dat.WrData[5] = 0x02;



42        iic_dat.WrData[6] = colorSignature;



43        iic_dat.WrData[7] = 0xFF;



44    } else



45    {



46        // Line tracking 



47        iic_dat.RdLng = 0x1C; 



48        iic_dat.WrLng = 0x08;  



49        iic_dat.WrData[4] = 0x30;



50        iic_dat.WrData[5] = 0x02;



51        iic_dat.WrData[6] = 0x00; 



52        iic_dat.WrData[7] = 0x01;     



53    }



54        



55    // Setup I2C communication



56    ioctl(iicFile, IIC_SETUP, &iic_dat);



57}



58



59// Initialize sensor



60void initCMUCam5Pixy2VisionSensor()



61{



62    iicFile = open(IIC_DEVICE_NAME, O_RDWR | O_SYNC);



63    



64    if (iicFile == -1) {



65        _EV3_DEBUG_PRINTF(("Failed to open vision sensor device file.\n"));



66        return;



67    } else {



68        pIic = (IIC*)mmap(0, sizeof(IIC), PROT_READ | PROT_WRITE, MAP_FILE | MAP_SHARED, iicFile, 0);



69        if (pIic == MAP_FAILED) {



70            _EV3_DEBUG_PRINTF(("Failed to map vision sensor memory.\n"));



71            return;



72        }



73    }



74    



75}

78// Get sensor values



79void getCMUCam5Pixy2VisionSensorValues(uint8_T portNumber, uint8_T i2cAddress, uint8_T mode, uint8_T sampleTime, uint8_T colorSignature, uint16_T *featureData, uint16_T *sigOutput, int16_T *angle)



80{



81    // Initialization



82    uint8_T Port = portNumber - 1;



83    uint8_T iscolormode = (mode==0) || (mode==1) || (mode==2);



84    uint8_T i = 0;



85    for(i = 0; i < 5; i++)



86    {



87        featureData[i] = 0;



88    }



89    *sigOutput = 0;



90    *angle = 0;



91    uint16_T sigMax = 65535;



92    if(mode==0) {



93        sigMax = 7;



94    }



95        



96    setIICCommand(Port, i2cAddress, mode, sampleTime, colorSignature);



97



98



99    uint8_T readlength = iic_dat.RdLng;



100    uint8_T syncbytelower = pIic->Raw[Port][pIic->Actual[Port]][readlength - 1];



101    uint8_T syncbyteupper = pIic->Raw[Port][pIic->Actual[Port]][readlength - 2];



102    uint8_T packettype = pIic->Raw[Port][pIic->Actual[Port]][readlength - 3];



103    uint8_T payloadlength = pIic->Raw[Port][pIic->Actual[Port]][readlength - 4];



104    



105   



106    // Check for sync bytes 



107    if( (syncbyteupper==UPPERSYNCBYTE) && (syncbytelower==LOWERSYNCBYTE) )



108    {



109        if( (iscolormode==1) && (packettype==33))



110        {



111            uint8_T blockx1 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 9];



112            uint8_T blockx2 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 10];



113            featureData[0] = validateCMUCam5Pixy2VisionSensorValues(blockx2*256+blockx1, 0, COLORTRACKINGFRAMEWIDTH);



114            _EV3_DEBUG_PRINTF((" BlockX is %u \n", featureData[0]));



115            uint8_T blocky1 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 11];



116            uint8_T blocky2 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 12];



117            featureData[1] = validateCMUCam5Pixy2VisionSensorValues(blocky2*256+blocky1, 0, COLORTRACKINGFRAMEHEIGHT);



118            _EV3_DEBUG_PRINTF((" BlockY is %u \n", featureData[1]));



119            uint8_T blockw1 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 13];



120            uint8_T blockw2 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 14];



121            featureData[2] = validateCMUCam5Pixy2VisionSensorValues(blockw2*256+blockw1, 0, COLORTRACKINGFRAMEWIDTH);



122            _EV3_DEBUG_PRINTF((" BlockW is %u \n", featureData[2]));



123            uint8_T blockh1 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 15];



124            uint8_T blockh2 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 16];



125            featureData[3] = validateCMUCam5Pixy2VisionSensorValues(blockh2*256+blockh1, 0, COLORTRACKINGFRAMEHEIGHT);



126            _EV3_DEBUG_PRINTF((" BlockH is %u \n", featureData[3]));



127            featureData[4] = payloadlength/BYTESPERDETECTEDBLOCK; 



128            _EV3_DEBUG_PRINTF((" Block Count is %u \n", featureData[4]));



129            uint8_T siglower = pIic->Raw[Port][pIic->Actual[Port]][readlength - 7];



130            uint8_T sigupper = pIic->Raw[Port][pIic->Actual[Port]][readlength - 8];



131            *sigOutput = convertDecimalToOctal(validateCMUCam5Pixy2VisionSensorValues(sigupper*256+siglower, 0, sigMax));



132            _EV3_DEBUG_PRINTF((" Signature is %u \n", *sigOutput));



133            uint8_T anglelower = pIic->Raw[Port][pIic->Actual[Port]][readlength - 17];



134            uint8_T angleupper = pIic->Raw[Port][pIic->Actual[Port]][readlength - 18];



135            *angle = angleupper*256+anglelower;



136            _EV3_DEBUG_PRINTF((" Angle is %u \n", *angle));



137            



138        } else if ( (iscolormode==0) && (packettype==49) )



139            {



140                uint8_T x0 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 9];



141                featureData[0] = validateCMUCam5Pixy2VisionSensorValues(x0, 0, LINETRACKINGFRAMEWIDTH);



142                _EV3_DEBUG_PRINTF((" x0 is %u \n", featureData[0]));



143                uint8_T y0 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 10];



144                featureData[1] = validateCMUCam5Pixy2VisionSensorValues(y0, 0, LINETRACKINGFRAMEHEIGHT);



145                _EV3_DEBUG_PRINTF((" y0 is %u \n", featureData[1]));



146                uint8_T x1 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 11];



147                featureData[2] = validateCMUCam5Pixy2VisionSensorValues(x1, 0, LINETRACKINGFRAMEWIDTH);



148                _EV3_DEBUG_PRINTF((" x1 is %u \n", featureData[2]));



149                uint8_T y1 = pIic->Raw[Port][pIic->Actual[Port]][readlength - 12];



150                featureData[3] = validateCMUCam5Pixy2VisionSensorValues(y1, 0, LINETRACKINGFRAMEHEIGHT);



151                _EV3_DEBUG_PRINTF((" y1 is %u \n", featureData[3]));



152                uint8_T flags = pIic->Raw[Port][pIic->Actual[Port]][readlength - 14];



153                if (flags == 4) {



154                    featureData[4] = 1;



155                } else {



156                    featureData[4] = 0;



157                }



158                _EV3_DEBUG_PRINTF((" Intersection Flag is %u \n", featureData[4]));



159            } else



160            {



161                _EV3_DEBUG_PRINTF(("Invalid data received from the Vision sensor.\n"));



162            }



163    }



164      



165}

167// Terminate sensor



168void terminateCMUCam5Pixy2VisionSensor(uint8_T portNumber)



169{



170    close(iicFile);



171    iicFile = -1;



172}



173



174// Validate the sensor values, return zero for invalid values



175uint16_T validateCMUCam5Pixy2VisionSensorValues(uint16_T value, uint16_T min, uint16_T max) 



176{



177    uint16_T sensorvalue = 0; 



178    if( (value >= min) && (value <= max)) {



179        sensorvalue = value;



180    }



181    return sensorvalue;



182}



183



184// Convert decimal signature value to octal



185uint16_T convertDecimalToOctal(uint16_T decimalNumber)



186{



187    uint16_T result = decimalNumber;



188    if(decimalNumber>7)



189    {



190        uint16_T octalNumber = 0;



191        uint16_T i = 1;



192        while (decimalNumber != 0)



193        {



194            octalNumber += (decimalNumber % 8) * i;



195            decimalNumber /= 8;



196            i *= 10;



197        }



198        result = octalNumber;



199    }



200    return result;
