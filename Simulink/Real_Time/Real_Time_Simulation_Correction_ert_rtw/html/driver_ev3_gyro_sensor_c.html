7#include "driver_ev3.h"

9



10extern int uartFile;



11extern UART *pUart;



12extern DEVCON DevCon;



13



14// Initialize sensor



15void initGyroSensor(uint8_T portNumber)



16{



17    uint8_T PORT = portNumber - 1;



18    



19    if(uartFile == -1) {



20        uartFile = open(UART_DEVICE_NAME, O_RDWR | O_SYNC);



21        if(uartFile == -1) {



22            _EV3_DEBUG_PRINTF(("Failed to open color sensor device file.\n"));



23            return;



24        }



25    }



26    



27    if(pUart == NULL) {



28        pUart = (UART*)mmap(0, sizeof(UART), PROT_READ | PROT_WRITE, MAP_FILE | MAP_SHARED, uartFile, 0);



29        if(pUart == MAP_FAILED) {



30            _EV3_DEBUG_PRINTF(("Failed to map color sensor memory.\n"));



31            return;



32        }



33    }



34    



35    DevCon.Mode[PORT] = 1;



36    DevCon.Connection[PORT] = CONN_INPUT_UART;



37    DevCon.Type[PORT] = 32;



38    



39    _EV3_DEBUG_PRINTF(("Gyro sensor initialized successfully on port %d.\n", portNumber));



40}



41



42// Get sensor value



43int32_T getGyroSensorValue(uint8_T portNumber)



44{



45    uint8_T PORT = portNumber - 1;



46    uint8_T rawData[2];



47    int32_T rawResult;



48    



49    rawData[0] = pUart->Raw[PORT][pUart->Actual[PORT]][0];



50    rawData[1] = pUart->Raw[PORT][pUart->Actual[PORT]][1];



51    rawResult = *((int16_T*)(&rawData[0]));



52    



53    _EV3_DEBUG_PRINTF(("gyro sensor on %d port - %d %d - %d - %d\n", portNumber, rawData[0], rawData[1], rawResult, -rawResult));



54    



55    return -rawResult;



56}



57



58// Terminate sensor



59void terminateGyroSensor(uint8_T portNumber)



60{



61    close(uartFile);



62    uartFile = -1;



63}
