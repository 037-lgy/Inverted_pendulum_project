7#include "driver_ev3.h"



8



9static int buttonFile = -1;



10static UI *pUI = NULL;



11



12// Initialize button device



13void initButton(uint8_T buttonOption)



14{



15    if (buttonFile == -1) {



16        buttonFile = open(UI_DEVICE_NAME, O_RDWR | O_SYNC);



17        if(buttonFile == -1) {



18            _EV3_DEBUG_PRINTF(("Failed to open button device file.\n"));



19            return;



20        }



21    }



22    



23    if (pUI == NULL) {



24        pUI = (UI*)mmap(0, sizeof(UI), PROT_READ | PROT_WRITE, MAP_FILE | MAP_SHARED, buttonFile, 0);



25        if(pUI == MAP_FAILED) {



26            _EV3_DEBUG_PRINTF(("Failed to map button memory.\n"));



27            return;



28        }



29    }



30    



31    _EV3_DEBUG_PRINTF(("Successfully initialized button device.\n"));



32}



33



34// Get button value



35uint8_T getButtonValue(uint8_T buttonOption)



36{



37    _EV3_DEBUG_PRINTF(("Button %d status - %d\n", buttonOption, pUI->Pressed[buttonOption - 1]));



38    



39    return pUI->Pressed[buttonOption - 1];



40}



41



42// Terminate button



43void terminateButton(uint8_T buttonOption)



44{



45    if (buttonFile != -1) {



46        close(buttonFile);



47        buttonFile = -1;



48    }



49    



50    _EV3_DEBUG_PRINTF(("Terminate button device.\n"));



51    



52}
