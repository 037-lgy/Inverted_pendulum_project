120bool SoundInitialized()



121{



122    return (SoundInstance.pSound != NULL) &&



123            (SoundInstance.pSound != &SoundInstance.Sound);



124}

126int WriteToSoundDevice(char * bytes, int num_bytes)



127{



128    ssize_t result = -1;



129    int sndHandle = open(LMS_SOUND_DEVICE_NAME, O_WRONLY);



130    if (sndHandle >= 0)



131    {



132        // for some reason write is not returning num_bytes -



133        // it usually returns zero



134        result = write(sndHandle, bytes, num_bytes);



135//printf("result = %zd\n", result);



136        close(sndHandle);



137//    if (result >= 0)



138//      return num_bytes;



139    }



140    return (int)result;



141}



142



143void SoundCloseDevices()



144{



145    if (!SoundInitialized())



146        return;



147    



148    if ((SoundInstance.pSound != NULL) &&



149            (SoundInstance.pSound != &SoundInstance.Sound))



150    {



151        munmap(SoundInstance.pSound, sizeof(unsigned short));



152    }



153    



154    SoundInstance.pSound = NULL;



155}



156



157bool SoundInit()



158{



159    int sndHandle;



160    bool Result = SoundInitialized();



161    if (!Result)



162    {



163        SoundInstance.SoundDriverDescriptor = -1;



164        SoundInstance.hSoundFile = -1;



165        SoundInstance.pSound = &SoundInstance.Sound;



166        



167        // Create a Shared Memory entry for signaling the driver state BUSY or NOT BUSY



168        



169        sndHandle = open(LMS_SOUND_DEVICE_NAME, O_RDWR | O_SYNC);



170        if (sndHandle != -1)



171        {



172            SOUND_BUSY * pTmp = (SOUND_BUSY*)mmap(0, sizeof(unsigned short),



173                    PROT_READ | PROT_WRITE, MAP_FILE | MAP_SHARED, sndHandle, 0);



174            if (pTmp == MAP_FAILED)



175            {



176                SoundCloseDevices();



177//        LogErrorNumber(SOUND_SHARED_MEMORY);



178                close(sndHandle);



179                // result is already false



180            }



181            else



182            {



183                SoundInstance.pSound = pTmp;



184                close(sndHandle);



185                Result = TRUE;



186            }



187        }



188    }



189    return Result;



190}

213void PlayToneEx(unsigned short frequency, unsigned short duration, byte volume)



214{



215    if (!SoundInitialized())



216        return;



217    



218    // sound system must not be muted to play a sound



219    if (SoundInstance.SoundMuted != 0)



220        return;



221    



222    (*SoundInstance.pSound).Busy = TRUE;



223    byte SoundData[6];



224    SoundData[0] = SOUND_CMD_TONE;



225    //SoundData[1] = (byte)(volume);



226    SoundData[1] = (byte)((volume*13)/100);



227    SoundData[2] = (byte)(frequency);



228    SoundData[3] = (byte)(frequency >> 8);



229    SoundData[4] = (byte)(duration);



230    SoundData[5] = (byte)(duration >> 8);



231    SoundInstance.SoundState = SOUND_STATE_TONE;



232    WriteToSoundDevice(SoundData, sizeof(SoundData)); // write 6 bytes



233}



234



235int SoundState()



236{



237    if (SoundInitialized())



238        return SoundInstance.SoundState;



239    return SOUND_STATE_IDLE;



240}



241



242void StopSound()



243{



244    if (!SoundInitialized())



245        return;



246    byte cmd = SOUND_CMD_BREAK;



247    SoundInstance.SoundState = SOUND_STATE_IDLE;



248    if (SoundInstance.hSoundFile >= 0)



249    {



250        close(SoundInstance.hSoundFile);



251        SoundInstance.hSoundFile  = -1;



252    }



253    WriteToSoundDevice(&cmd, 1);



254}



255



256void MuteSound()



257{



258    // first stop any playing sounds



259    StopSound();



260    SoundInstance.SoundMuted = 1;



261}



262



263void UnmuteSound()



264{



265    SoundInstance.SoundMuted = 0;



266}



267



268void ClearSound()



269{



270    // a synonym for StopSound;



271    StopSound();



272}
