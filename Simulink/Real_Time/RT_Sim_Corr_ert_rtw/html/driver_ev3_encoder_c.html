9static int encoderFile = -1;



10static MOTORDATA *pEncoder;



11



12// Initialize encoder



13void initEncoder(uint8_T portNumber)



14{



15    if(encoderFile == -1) {



16        encoderFile = open(MOTOR_DEVICE_NAME, O_RDWR | O_SYNC);



17        if(encoderFile == -1) {



18            _EV3_DEBUG_PRINTF(("Failed to open encoder device file.\n"));



19            exit(1);



20        }



21        pEncoder = (MOTORDATA*)mmap(0, sizeof(MOTORDATA)*vmOUTPUTS, PROT_READ | PROT_WRITE, MAP_FILE | MAP_SHARED, encoderFile, 0);



22        if(pEncoder == MAP_FAILED) {



23            _EV3_DEBUG_PRINTF(("Failed to map encoder memory.\n"));



24            exit(1);



25        }



26        _EV3_DEBUG_PRINTF(("Successfully initialize encoder - %d - %d.\n", encoderFile, pEncoder));



27    }



28}



29



30// Get encoder value



31int32_T getEncoderValueNoReset(uint8_T portNumber)



32{



33    uint8_T PORT = portNumber - 1;



34    int32_T rawResult;



35    



36    rawResult = pEncoder[PORT].TachoSensor;



37    



38    _EV3_DEBUG_PRINTF(("No Reset: Encoder on %d output port - %d %d %d %d\n", portNumber, pEncoder[PORT].Speed, pEncoder[PORT].TachoCounts, pEncoder[PORT].TachoSensor, rawResult));



39    



40    return rawResult;



41}



42



43int32_T getEncoderValueSampleReset(uint8_T portNumber)



44{



45    uint8_T PORT = portNumber - 1;



46    static int32_T oldResult[4];



47    int32_T newResult;



48    int32_T result;



49    



50    newResult = pEncoder[PORT].TachoSensor;



51    result = newResult - oldResult[PORT];



52    oldResult[PORT] = newResult;



53    



54    _EV3_DEBUG_PRINTF(("Sample Reset: Encoder on %d output port - %d %d %d - %d\n", portNumber, pEncoder[PORT].Speed, pEncoder[PORT].TachoCounts, pEncoder[PORT].TachoSensor, result));



55    return result;



56}



57



58int32_T getEncoderValueExternalReset(uint8_T portNumber, int8_T *in)



59{



60    uint8_T PORT = portNumber - 1;



61    static int32_T oldResult[4];



62    int32_T newResult;



63    int32_T result;



64    



65    newResult = pEncoder[PORT].TachoSensor;



66    result = newResult - oldResult[PORT];



67    if(*in != 0) {



68        oldResult[PORT] = newResult;



69    }



70    



71    _EV3_DEBUG_PRINTF(("External Reset: Encoder on %d output port - %d %d %d - %d\n", portNumber, pEncoder[PORT].Speed, pEncoder[PORT].TachoCounts, pEncoder[PORT].TachoSensor, result));



72    return result;



73}



74



75// Terminate encoder



76void terminateEncoder(uint8_T portNumber)



77{



78    if(encoderFile != -1) {



79        close(encoderFile);



80        encoderFile = -1;



81        _EV3_DEBUG_PRINTF(("Closing encoder file.\n"));



82    }



83}
